def extract_mfg_mapping_table_strict(self, raw_bytes: bytes, setup_pos: int) -> list:
    """
    ПРАВКА №2: Строгий побайтовый прыжок по 16-битному LE/BE указателю.
    Защищает от Out of bounds и бинарного мусора выравнивания.
    """
    window = raw_bytes[setup_pos:setup_pos + 2048]
    key = b'id_map_info_ptr\x00'
    idx = window.find(key)
    if idx == -1:
        return []
    
    # Смещение лежит СТРОГО за терминатором \x00 и занимает ровно 2 байта (uint16)
    off_pos = idx + len(key)
    fmt = f"{self.endian}H" # Динамический эндиан (правка №1)
    
    rel_offset = struct.unpack(fmt, window[off_pos:off_pos+2])[0]
    table_start = setup_pos + rel_offset
    
    # Буфер таблицы считывается до следующего маркера setup_data
    next_setup = raw_bytes.find(b'setup_data', table_start)
    table_end = next_setup if next_setup != -1 else min(len(raw_bytes), table_start + 4096)
    
    return self._parse_flat_strings(raw_bytes[table_start:table_end])
Используйте код с осторожностью.

• Решение: Считываем первые байты UGC-заголовка, определяем платформу и на лету меняем маску struct (< на >).

def _detect_architecture_endian(self, file_content: bytes) -> str:
    """
    ПРАВКА №1: Автоопределение байтового порядка хоста.
    Добавьте этот вызов в метод открытия файла перед распаковкой TOC.
    """
    # Сканируем первые 500 байт UGC-заголовка
    header_chunk = file_content[:500]
    if b"SUN_" in header_chunk or b"SunOS" in header_chunk:
        self.endian_mask = ">"  # Переключаемся на Big-Endian (SunOS)
        return "Big-Endian"
    else:
        self.endian_mask = "<"  # Little-Endian (Windows по умолчанию)
        return "Little-Endian"

def extract_mfg_mapping_table_clean(self, raw_bytes: bytes, setup_pos: int) -> list:
    """
    ПРАВКА №2: Извлечение технологических связей Mapping Table.
    Заменяет нестабильный поиск через регулярные выражения.
    """
    # Ограничиваем буфер поиска рамками текущей структуры setup_data
    window = raw_bytes[setup_pos:setup_pos + 2048]
    key = b'id_map_info_ptr\x00'
    idx = window.find(key)
    if idx == -1:
        return []
    
    # Смещение находится строго за маркером и занимает ровно 2 байта (uint16)
    offset_pos = idx + len(key)
    if offset_pos + 2 > len(window):
        return []
        
    # Распаковываем с использованием маски из Правки №1 (self.endian_mask)
    unpack_fmt = f"{self.endian_mask}H"
    rel_offset = struct.unpack(unpack_fmt, window[offset_pos:offset_pos+2])[0]
    
    # Вычисляем абсолютный адрес начала таблицы маппинга в файле
    table_start = setup_pos + rel_offset
    if table_start >= len(raw_bytes):
        return []
        
    # Считываем буфер до начала следующего блока параметров Creo
    next_setup = raw_bytes.find(b'setup_data', table_start)
    table_end = next_setup if next_setup != -1 else min(len(raw_bytes), table_start + 4096)
    
    # Извлекаем чистые latin-1/ascii строки (ID станков, инструментов, приспособлений)
    raw_table_data = raw_bytes[table_start:table_end]
    return [match.group().decode('latin-1', errors='ignore').strip() 
            for match in re.finditer(rb'[\x20-\x7e]{3,64}', raw_table_data)]
