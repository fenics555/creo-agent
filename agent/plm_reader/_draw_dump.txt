function draw(root){
  const N = {0: [root]}, L = [];
  // ДВА НЕЗАВИСИМЫХ ОБХОДА: вниз (потомки) и вверх (предки).
  // Раньше был один проход — он уходил только вниз и «вверх» не рисовался никогда.
  function go(start, dir){
    const seen = new Set([root]);
    let frontier = [start];
    for (let depth = 1; depth <= 3 && frontier.length; depth++){
      const next = [];
      for (const m of frontier){
        // ЖЁСТКИЙ ЛИМИТ: иначе на базе (26k+ узлов) хаб с сотнями детей
        // переполняет обход и граф не рисуется вовсе
        if (Object.keys(N).length > 120) return;
        const n = ND[m]; if (!n) continue;
        const use = (dir === "d" ? n.o : n.n) || [];
        for (const t of use.slice(0, 25)){
          const nm = t[0];
          if (!ND[nm] || nm === root || seen.has(nm)) continue;
          seen.add(nm);
          const lvl = depth * dir;
          (N[lvl] = N[lvl] || []).push(nm);
          const role = edgeKind(dir === "d" ? m : nm, dir === "d" ? nm : m);
          const main = (depth === 1);
          const k = (role === "base" || role === "copy") ? role
                  : (main ? "main-" + dir : "deep-" + dir);
          // ребро в данных ND всегда «родитель → ребёнок»:
          const eFrom = dir === "d" ? m : nm, eTo = dir === "d" ? nm : m;
          L.push([eFrom, eTo, k]);
          next.push(nm);
        }
      }
      frontier = next;
    }
  }
  go(root, "d");   // вниз: main-down (1-й уровень), deep-down (глубже)
  go(root, "u");   // вверх: main-up (1-й уровень), deep-up (глубже)

  const rows = {};
  Object.entries(N).forEach(([m, d]) => { (rows[d] = rows[d] || []).push(m); });
  const cols = Object.keys(rows).map(Number).sort((a, b) => a - b);
  const CW = 250, RH = 24, pos = {};
  let x = 20;
  for (const c of cols){
    rows[c].sort(); let y = 20;
    for (const m of rows[c]){ pos[m] = [x, y]; y += RH; }
    x += CW;
  }
  const W = Math.max(500, cols.length * CW + 40);
  const H = Math.max(200, Math.max(...cols.map(c => rows[c].length)) * RH + 50);
  let svg = "<defs>";
  const used = {};
  for (const [a, b, k] of L){ used[k] = (used[k] || 0) + 1; }
  for (const k of Object.keys(CLR)){
    svg += '<marker id="ar-' + k + '" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6" ' +
           'markerHeight="6" orient="auto"><path d="M0 0 L10 5 L0 10 z" fill="' +
           CLR[k].s + '"/></marker>';
  }
  svg += "</defs>";
  for (const [a, b, k] of L){
    const p = pos[a], r = pos[b]; if (!p || !r) continue;
    const x1 = p[0] + 212, y1 = p[1] + 9, x2 = r[0], y2 = r[1] + 9;
    const c = (CLR[k] || CLR["deep-down"]).s;
    svg += '<path d="M' + x1 + ' ' + y1 + ' C' + ((x1+x2)/2) + ' ' + y1 + ' ' +
           ((x1+x2)/2) + ' ' + y2 + ' ' + x2 + ' ' + y2 +
           '" stroke="' + c + '" stroke-width="' + (k.startsWith("main") ? "2" : "1.2") +
           '" fill="none" marker-end="url(#ar-' + k + ')"/>';
  }
  for (const m of Object.keys(pos)){
    const [px, py] = pos[m], n = ND[m];
    const c = m === root ? "#56a6ff" : (n.m ? "#c98a5a" : "#59666f");
    svg += '<g class="nd" data-m="' + m + '" style="cursor:pointer">' +
      '<rect x="' + px + '" y="' + py + '" width="212" height="18" rx="4" fill="' +
      (m === root ? "#213042" : "#1b2128") + '" stroke="' + c + '"/>' +
      '<text x="' + (px + 6) + '" y="' + (py + 13) + '">' + m.slice(0, 32) + '</text></g>';
  }
  let leg = "";
  for (const k of Object.keys(CLR)){
    if (!used[k]) continue;
    leg += '<span style="display:inline-flex;align-items:center;gap:5px;margin-right:14px">' +
           '<svg width="22" height="8"><line x1="0" y1="4" x2="22" y2="4" stroke="' +
           CLR[k].s + '" stroke-width="' + (k.startsWith("main") ? "2.5" : "1.5") +
           '"/></svg>' + CLR[k].t + ' <b style="color:#8fa0b3">(' + used[k] + ')</b></span>';
  }
  document.getElementById("graph").innerHTML =
    '<div style="padding:8px 12px;color:#8fa0b3;font-size:11px;border-bottom:1px solid #20262c">' +
    'граф: ' + Object.keys(N).length + ' узлов · ' + L.length +
    ' связей (клик по узлу — перейти)<div style="margin-top:5px">' + leg + '</div></div>' +
    '<svg width="' + W + '" height="' + H + '">' + svg + '</svg>';
  document.querySelectorAll("#graph .nd").forEach(el =>
    el.addEventListener("click", () => show(el.dataset.m)));
}
